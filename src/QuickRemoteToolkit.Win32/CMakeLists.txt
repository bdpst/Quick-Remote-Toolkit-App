cmake_minimum_required(VERSION 3.20)
project(QuickRemoteToolkitWin32 LANGUAGES CXX RC)
add_executable(QuickRemoteToolkitWin32 WIN32 main.cpp app.rc)
target_compile_features(QuickRemoteToolkitWin32 PRIVATE cxx_std_17)
target_compile_definitions(QuickRemoteToolkitWin32 PRIVATE UNICODE _UNICODE)
target_link_libraries(QuickRemoteToolkitWin32 PRIVATE comctl32 comdlg32 shell32 gdi32 user32)
if(MINGW)
    target_link_options(QuickRemoteToolkitWin32 PRIVATE -municode -static)
endif()
if(MSVC)
    target_compile_options(QuickRemoteToolkitWin32 PRIVATE /utf-8)
endif()
